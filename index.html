async function loadSnowWeather() { const alertBox = document.getElementById("snowAlert"); const currentBox = document.getElementById("currentWeather"); const forecastBox = document.getElementById("forecastGrid"); const updated = document.getElementById("weatherUpdated"); if (!alertBox || !currentBox || !forecastBox) { alert("Snow alert HTML section is missing. Check your element IDs."); return; } if (!currentCoords) { alertBox.className = "snow-alert error"; alertBox.textContent = "GPS location not found. Click Use My Current Location first."; return; } checkWeatherBtn.disabled = true; checkWeatherBtn.textContent = "Loading weather..."; alertBox.className = "snow-alert"; alertBox.textContent = "Connecting to live weather service..."; try { const params = new URLSearchParams({ latitude: currentCoords.lat, longitude: currentCoords.lon, current: "temperature_2m,relative_humidity_2m,apparent_temperature,snowfall,weather_code,wind_speed_10m", hourly: "temperature_2m,snowfall,precipitation_probability,weather_code", forecast_days: "2", timezone: "auto" }); const response = await fetch( "https://api.open-meteo.com/v1/forecast?" + params ); if (!response.ok) { throw new Error("Weather API error: " + response.status); } const data = await response.json(); if (!data.current || !data.hourly) { throw new Error("Weather data is missing."); } const now = data.current; const hourly = data.hourly; currentBox.innerHTML = `

${now.temperature_2m}°C

${weatherDescription(now.weather_code)}

Feels like: ${now.apparent_temperature}°C

Humidity: ${now.relative_humidity_2m}%

Wind: ${now.wind_speed_10m} km/h

Current snowfall: ${now.snowfall} cm

`; forecastBox.innerHTML = ""; const snowCodes = [71, 73, 75, 77, 85, 86]; const nowTime = new Date(now.time).getTime(); let snowExpected = false; let maxSnow = 0; let snowTime = ""; for (let i = 0; i < hourly.time.length; i++) { const forecastDate = new Date(hourly.time[i]); const hoursAhead = (forecastDate.getTime() - nowTime) / 3600000; if (hoursAhead < 0 || hoursAhead > 24) continue; const snow = Number(hourly.snowfall[i] || 0); const code = Number(hourly.weather_code[i]); if (snow > 0 || snowCodes.includes(code)) { snowExpected = true; if (snow > maxSnow) { maxSnow = snow; snowTime = forecastDate.toLocaleTimeString([], { hour: "numeric", minute: "2-digit" }); } } if (Math.round(hoursAhead) % 3 === 0) { const item = document.createElement("div"); item.className = "forecast-item"; item.textContent = forecastDate.toLocaleTimeString([], { hour: "numeric" }) + " | " + hourly.temperature_2m[i] + "°C | " + weatherDescription(code) + " | Snow: " + snow + " cm"; forecastBox.appendChild(item); } } const currentSnow = Number(now.snowfall || 0); const currentCode = Number(now.weather_code); if (currentSnow > 0 || snowCodes.includes(currentCode)) { alertBox.className = "snow-alert"; alertBox.textContent = "❄️ SNOWFALL ALERT! Snow is currently indicated at your location. " + "Current snowfall: " + currentSnow + " cm. Please check official road advisories."; sendSnowNotification("Snowfall is currently indicated at your location."); } else if (snowExpected) { alertBox.className = "snow-alert"; alertBox.textContent = "🌨️ SNOW WARNING! Snow is forecast in the next 24 hours. " + "Maximum hourly snowfall: " + maxSnow + " cm around " + snowTime + ". Please plan ahead."; sendSnowNotification("Snow is forecast in the next 24 hours."); } else { alertBox.className = "snow-alert safe"; alertBox.textContent = "✅ No snowfall is forecast for your location in the next 24 hours. " + "Temperature: " + now.temperature_2m + "°C. Check again before travelling."; } updated.textContent = "Last updated: " + new Date().toLocaleString(); } catch (error) { alertBox.className = "snow-alert error"; alertBox.textContent = "Weather error: " + error.message; } finally { checkWeatherBtn.disabled = false; checkWeatherBtn.textContent = "Check Live Snow Alert & Forecast"; } }
